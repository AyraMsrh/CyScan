  <script>
    const openParamModalBtn = document.getElementById('openParamModal');
    const paramModal = document.getElementById('paramModal');
    const cancelParam = document.getElementById('cancelParam');
    const scanForm = document.getElementById('scanForm');
    const scanOverlay = document.getElementById('scanModal');
    const scanIp = document.getElementById('scanIp');
    const portModeRadios = document.querySelectorAll('input[name="port_mode"]');
    const customPortsWrap = document.getElementById('customPortsWrap');

    // open parameter modal
    openParamModalBtn?.addEventListener('click', () => {
      paramModal.classList.remove('hidden');
    });

    // close parameter modal
    const closeParamModal = () => paramModal.classList.add('hidden');
    cancelParam?.addEventListener('click', closeParamModal);
    paramModal.addEventListener('click', (e) => {
      if (e.target === paramModal) closeParamModal();
    });
    window.addEventListener('keydown', (e) => {
      if (e.key === 'Escape') closeParamModal();
    });

    // toggle custom ports input visibility
    const updatePortMode = () => {
      const mode = [...portModeRadios].find(r => r.checked)?.value;
      if (mode === 'custom') customPortsWrap.classList.remove('hidden');
      else customPortsWrap.classList.add('hidden');
    };
    portModeRadios.forEach(r => r.addEventListener('change', updatePortMode));
    updatePortMode();

    // on submit: validate minimal things, show spinner overlay
    scanForm?.addEventListener('submit', (e) => {
      const formData = new FormData(scanForm);
      const ip = formData.get('ip')?.toString().trim();
      const portMode = formData.get('port_mode');

      if (!ip) {
        e.preventDefault();
        alert('Please enter an IP address.');
        return;
      }
      if (portMode === 'custom') {
        const custom = formData.get('custom_ports')?.toString().trim();
        if (!custom) {
          e.preventDefault();
          alert('Please provide a custom port list (e.g., 22,80,443 or 1-1024).');
          return;
        }
      }

      closeParamModal();
      if (scanIp && ip) scanIp.textContent = `Target: ${ip}`;
      scanOverlay.classList.remove('hidden');
    });

 
  {% if result %}
  document.addEventListener("DOMContentLoaded", () => {
    scanOverlay.classList.add('hidden');
    document.getElementById('resultModal').classList.remove('hidden');
  });
  {% endif %}

  // make "Request" button light up when clicked
  document.querySelectorAll('.request-btn').forEach(btn => {
    btn.addEventListener('click', () => {
      btn.classList.remove('bg-gray-200', 'hover:bg-gray-300');
      btn.classList.add('bg-yellow-400', 'text-black', 'font-semibold');
      btn.textContent = 'Requested';
    });
  });


  </script>
